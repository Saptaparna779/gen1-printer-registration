+CLOUD_ID_PATTERN = re.compile(r"^CID-[A-F0-9]{12}$")
+EMAIL_PATTERN = re.compile(r"^[a-z0-9]{10}@print\.hpeprint\.com$")
+CLAIM_CODE_PATTERN = re.compile(r"^[A-Z0-9]{8}$")

+
+def _register_printer(client, serial_number: str, model_number: str, firmware_version: str, simulate_welcome_page_failure: bool = False):
+    return client.post(
+        "/printers/register",
+        json={
+            "serial_number": serial_number,
+            "model_number": model_number,
+            "firmware_version": firmware_version,
+            "simulate_welcome_page_failure": simulate_welcome_page_failure,
+        },
+    )
+
+
+def _claim_printer(client, claim_code: str, user_id: str):
+    return client.post(
+        "/printers/claim",
+        json={"claim_code": claim_code, "user_id": user_id},
+    )
+
+
+def _get_printer(client, printer_id: str):
+    return client.get(f"/printers/{printer_id}")
+
+

+    """[HAPPY PATH] Re-register an already-claimed printer and confirm owner_user_id remains unchanged after re-registration."""
+    initial_register = _register_printer(
